// {{name}}: a shepherd plugin. Read AGENTS.md before changing it.
import { runPlugin, type Pane } from "./shepherd-plugin";

runPlugin(async (shepherd) => {
  // what this plugin knows: each pane's detected agent, keyed by pane instance
  const agents = new Map<string, NonNullable<Pane["agent"]>>();

  // offer actions to `shepherd plugin run {{name}} <action>`
  await shepherd.hello({
    status: () => ({ agents: Object.fromEntries(agents) }),
  });

  await shepherd.subscribe({
    // every pane as of now: this was already true when the plugin started, so it isn't news
    onSnapshot: (snapshot) => {
      for (const pane of snapshot.panes) if (pane.agent) agents.set(pane.instance, pane.agent);
    },
    // each later change, once, in order
    onEvent: (event) => {
      if (event.type !== "agent.state" || !event.instance) return;
      if (event.harness && event.to) agents.set(event.instance, { harness: event.harness, state: event.to, source: "screen" });
      else agents.delete(event.instance);
      // the user's logic goes here
    },
  });
  console.log("{{name}}: watching");
});
